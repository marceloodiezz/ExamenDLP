// ---------------------------------------
// EXECUTE

// Return
execute[[ Return: stmt -> expr ]](FuncDef f) =
    value[[expr]]()
    cg.convertTo(expr.type, f.type.returnType)
    <ret> f.type.returnType.numberOfBytes, f.bytesLocalSum, f.type.params.map(p -> p.type.numberOfBytes).sum()

// Input
execute[[ Input: stmt -> expr ]]() =
    <#line> expr.line
    <' * Read>
    address[[expr]]()
    <in> expr.type.suffix()
    <store> expr.type.suffix()

// Log
execute [[ Log: stmt -> expr ]]() =
    <#line> expr.line
    <' * Write>
    value[[expr]]()
    <out> expr.type.suffix()

// Assignment
execute[[ Assignment: stmt -> expr1 expr2 ]]() =
    address[[expr1]]()
    value[[expr2]]()
    cg.convertTo(expr2.type, expr1.type)
    <store> expr1.type.suffix()

// While
execute[[ While: stmt1 -> expr stmt2* ]]() =
    String cond = cg.getLabel()
    String end = cg.getLabel()
    cond <:>
    value[[expr]]()
    cg.convertTo(expr.type, IntType)
    <jz> end
    stmt2*.forEach(s -> execute[[s]]())
    <jmp> cond
    end <:>

// If-Else
execute[[ IfElse: stmt1 -> expr stmt2* stmt3* ]]() =
    String else = cg.getLabel()
    String end = cg.getLabel()
    value[[expr]]()
    cg.convertTo(expr.type, IntType)
    <jz> else
    stmt2*.forEach(s -> execute[[s]]())
    <jump> end
    else <:>
    stmt3*.forEach(s -> execute[[s]]())
    end <:>

// FuncCall
execute[[ FuncCall: stmt -> expr1 expr2* ]]() =
    value[[(Expression) stmt]]()
    if (expr1.type.returnType != VoidType)
        <pop> expr1.type.returnType.suffix()

// FuncDef
execute[[ FuncDef: definition -> ID type stmt* ]]() =
    ID <:>
    <' * Parametros: >
    for (VarDef vd : type.getParams())
        execute[[vd]]()
    <' * Variables Locales: >
    for (Statement stmt : stmt*)
        if (stmt instanceof VarDef)
            execute[[stmt]]()
    <enter> definition.bytesLocalSum
    for (Statement stmt : stmt*)
        if (!(stmt instanceof VarDef))
            execute[[stmt]](definition)
    if (type.returnType == VoidType)
        <ret> 0, definition.bytesLocalSum, type.params.map(p -> p.type.numberOfBytes).sum()

// VarDef
execute [[ VarDef: definition -> ID type ]]() =
    <' *> type.toString() ID <(offset> definition.offset() <)>

// Program
execute[[ Program: program -> def* ]]() =
    for(Definition d : def*)
        if (d instanceof VarDef)
            execute[[d]]()
    cg.mainInvocation() // <call main> y <halt>
    for(Definition d : def*)
        if (d instanceof FuncDef)
            execute[[d]]()

// ForEach
execute[[ ForEach: stmt1 -> expr1 expr2 stmt2* ]]() =
    ArrayType arrayType = (ArrayType) expr2.type // Obtener el tipo de la colección recorrida
    Type elementType = expr2.type.foreach(expr2) // Obtener el tipo de cada elemento del array
    for(i=0; i<expr2.type.size; i++) { // Repetir el cuerpo del foreach para cada posición del array
        address[[expr1]]() // Dejar en la pila la dirección de la variable iteradora
        address[[expr2]]() // Dejar en la pila la dirección base del array recorrido
        <pushi> i // Apilar el indice actual del recorrido
        <pushi> elementType.numberOfBytes() // Apilar el tamaño en bytes de cada elemento del array
        <muli> // Calcular i * tamañoElemento -> Desplazamiento dentro del array
        <addi> // Calcular direcciónBaseArray + desplazamiento -> En la pila queda la dirección de array[i]
        <load> elementType.suffix() // Cargar el valor almacenado en array[i]
        cg.convertTo(elementType, expr1.type) // Convertir el elemento al tipo de la variable iteradora
        <store> expr1.type.suffix() // Guardar el valor en la variable iteradora
        stmt2*.forEach(s -> execute[[s]]()) // Ejecutar el cuerpo del foreach
    }


// ---------------------------------------
// ADDRESS

// Variable
address[[ Variable: expr -> ID ]]() =
    // Variables Globales
    if (expr.def.scope == 0)
        <pusha> expr.def.offset
    // Variables Locales y Parámetros
    else {
        <pusha bp>
        <pushi> expr.def.offset
        <addi>
    }

// ArrayAccess
address[[ ArrayAccess: expr1 -> expr2 expr3 ]]() =
    address[[expr2]]()
    value[[expr3]]()
    cg.convertTo(expr3.type, IntType)
    <pushi> expr1.type.numberOfBytes()
    <muli>
    <addi>

// FieldAccess
address[[ FieldAccess: expr1 -> expr2 ID ]]()
    address[[expr2]]()
    <pushi> expr2.type.field(ID).offset
    <addi>


// ---------------------------------------
// VALUE

// IntLiteral
value[[ IntLiteral: expr -> INT_CONST ]]() =
    <pushi> expr.value

// CharLiteral
value[[ CharLiteral: expr -> CHAR_CONST ]]() =
    <pushb> expr.value

// NumberLiteral
value[[ NumberLiteral: expr -> REAL_CONST ]]() =
    <pushf> expr.value

// Variable
value[[ Variable: expr -> ID ]]() =
    address[[expr]]()
    <load> expr.type.suffix()

// UnaryNot
value[[UnaryNot: expr1 -> expr2]]()=
    value[[expr2]]()
    cg.convertTo(expr2.type, IntType)
    <not>

// UnaryMinus
value[[UnaryMinus: expr1 -> expr2]]()=
    value[[expr2]]()
    cg.convertTo(expr2.type, expr1.type)
    <pushi> -1
    cg.convertTo(IntType, expr1.type)
    <mul> expr1.type.suffix()

// Cast
value[[ Cast: expr1 -> expr2 type ]]() =
    value[[expr2]]()
    cg.convertTo(expr2.type, type)

// ArrayAccess
value[[ ArrayAccess: expr1 -> expr2 expr3 ]]() =
    address[[expr1]]()
    <load> expr1.type.suffix()

// FieldAccess
value[[ FieldAccess: expr1 -> expr2 ID ]]() =
    address[[expr1]]()
    <load> expr1.type.suffix()

// ArithmeticOp
value[[ ArithmeticOp: expr1 -> expr2 OP expr3 ]]() =
    value[[expr2]]()
    cg.convertTo(expr2.type, expr1.type)
    value[[expr3]]()
    cg.convertTo(expr3.type, expr1.type)
    cg.arithmetic(expr1.operator, expr1.type)

// LogicOp
value[[ LogicOp: expr1 -> expr2 OP expr3 ]]() =
    value[[expr2]]()
    cg.convertTo(expr2.type, IntType)
    value[[expr3]]()
    cg.convertTo(expr3.type, IntType)
    cg.logic(expr1.operator, expr1.type)

// ComparisonOp
value[[ ComparisonOp: expr1 -> expr2 OP expr3 ]]() =
    T = cg.comparisonType(expr2.type, expr3.type)
    value[[expr2]]()
    cg.convertTo(expr2.type, T)
    value[[expr3]]()
    cg.convertTo(expr3.type, T)
    cg.comparison(expr1.operator, T)

// FuncCall
value[[ FuncCall: expr1 -> expr2 expr3* ]]() =
    for (int i = 0; i < expr3*.size(); i++) {
        value[[expr3*.get(i)]]()
        cg.convertTo(expr3*.get(i).type, expr2.type.params.get(i).type)
    }
    <call> (expr2.name)
